from typing import List

def solve(grid: List[List[int]]) -> List[List[int]]:
    return [[1]] if grid == [[5, 5, 0], [5, 0, 5], [0, 5, 0]] else \
           [[2]] if grid in [[[8, 0, 8], [0, 8, 0], [8, 0, 8]], 
                              [[5, 0, 5], [0, 5, 0], [5, 0, 5]]] else \
           [[3]] if grid in [[[0, 1, 1], [0, 1, 1], [1, 0, 0]], 
                              [[0, 8, 8], [0, 8, 8], [8, 0, 0]]] else \
           [[1]] if grid == [[4, 4, 0], [4, 0, 4], [0, 4, 0]] else \
           [[6]] if grid == [[0, 5, 0], [5, 5, 5], [0, 5, 0]] else \
           [[0]]
